42 @staticmethod 

43 def _to_task(obj: Any, calendar_id: str = "") -> Task: 

44 """Convert a caldav VTODO object to our :class:`Task`. 

45 

46 Reads via caldav 2.0's ``icalendar_component`` (the ``icalendar`` lib), 

47 same pattern as ``_to_calendar_event`` but for VTODO fields. 

48 """ 

49 comp = obj.icalendar_component 

50 

51 return Task( 

52 uid=_comp_text(comp, "UID"), 

53 summary=_comp_text(comp, "SUMMARY"), 

54 description=_comp_text(comp, "DESCRIPTION"), 

55 dtstart=_comp_dt(comp, "DTSTART"), 

56 due=_comp_dt(comp, "DUE"), 

57 status=_comp_text(comp, "STATUS"), 

58 calendar_id=calendar_id, 

59 )

61 def _task_to_ical(self, task: Task) -> str: 

62 """Build an iCalendar string from a :class:`Task`.""" 

63 import datetime 

64 

65 e = self._escape_text 

66 dtstamp = datetime.datetime.now(datetime.UTC).strftime("%Y%m%dT%H%M%SZ") 

67 ical = ( 

68 "BEGIN:VCALENDAR\n" 

69 "VERSION:2.0\n" 

70 "PRODID:-//robotsix-calendar-agent//EN\n" 

71 "BEGIN:VTODO\n" 

72 f"UID:{task.uid or ''}\n" 

73 f"DTSTAMP:{dtstamp}\n" 

74 f"SUMMARY:{e(task.summary)}\n" 

75 f"DESCRIPTION:{e(task.description)}\n" 

76 ) 

77 if task.dtstart: 

78 ical += f"{self._ical_dt('DTSTART', task.dtstart)}\n" 

79 if task.due: 

80 ical += f"{self._ical_dt('DUE', task.due)}\n" 

81 if task.status: 

82 ical += f"STATUS:{task.status}\n" 

83 ical += "END:VTODO\nEND:VCALENDAR\n" 

84 return ical

86 def _find_task_by_uid(self, uid: str) -> tuple[Any, Any] | None: 

87 """Locate a task by UID across all calendars. 

88 

89 Returns ``(calendar, task_obj)`` or ``None`` if not found. 

90 """ 

91 for cal in self._iter_calendars(""): 

92 try: 

93 task_obj = cal.get_todo_by_uid(uid) 

94 return cal, task_obj 

95 except self._caldav.lib.error.NotFoundError: 

96 continue 

97 return None

103 @_wrap_caldav_op("list tasks") 

104 def list_tasks(self, calendar_id: str = "") -> list[Task]: 

105 """Return all VTODO tasks from CalDAV calendar collections. 

106 

107 When *calendar_id* is empty, tasks are aggregated from **all** 

108 calendars. Each task is tagged with its source ``calendar_id``. 

109 """ 

110 logger.debug("list_tasks calendar_id=%r", calendar_id) 

111 aggregated: list[Task] = [] 

112 for cal in self._iter_calendars(calendar_id): 

113 results = cal.search(todo=True) 

114 aggregated.extend(self._to_task(r, calendar_id=cal.name) for r in results) 

115 return aggregated

117 @_wrap_caldav_op("create task") 

118 def create_task(self, task: Task, calendar_id: str = "") -> Task: 

119 """Create a VTODO task; return the task with its server-assigned uid. 

120 

121 If *calendar_id* is empty, use the default calendar. 

122 """ 

123 logger.debug( 

124 "create_task uid=%r calendar_id=%r summary=%r", 

125 task.uid, 

126 calendar_id, 

127 task.summary, 

128 ) 

129 if not task.uid: 

130 task = Task( 

131 uid=str(uuid.uuid4()), 

132 summary=task.summary, 

133 description=task.description, 

134 dtstart=task.dtstart, 

135 due=task.due, 

136 status=task.status, 

137 calendar_id=task.calendar_id, 

138 ) 

139 cal = self._get_calendar(calendar_id) 

140 ical = self._task_to_ical(task) 

141 saved = cal.save_todo(ical) 

142 return self._to_task(saved, calendar_id=cal.name)

144 @_wrap_caldav_op("update task") 

145 def update_task(self, uid: str, task: Task, calendar_id: str = "") -> Task: 

146 """Update the task identified by *uid*; return the updated task. 

147 

148 When *calendar_id* is empty, iterates **all** calendars to locate 

149 the UID (the UID may live in a non-default collection). When 

150 *calendar_id* is given, only that single calendar is searched. 

151 

152 Raises: 

153 NotFoundError: If the UID doesn't exist. 

154 """ 

155 logger.debug( 

156 "update_task uid=%r calendar_id=%r summary=%r", 

157 uid, 

158 calendar_id, 

159 task.summary, 

160 ) 

161 if calendar_id: 

162 cal = self._get_calendar(calendar_id) 

163 cal.get_todo_by_uid(uid) 

164 else: 

165 result = self._find_task_by_uid(uid) 

166 if result is None: 

167 raise NotFoundError( 

168 f"Task with UID {uid!r} not found.", 

169 ) 

170 cal, _ = result 

171 # Build updated iCal with the same UID 

172 updated = Task( 

173 uid=uid, 

174 summary=task.summary, 

175 description=task.description, 

176 dtstart=task.dtstart, 

177 due=task.due, 

178 status=task.status, 

179 calendar_id=calendar_id or cal.name, 

180 ) 

181 ical = self._task_to_ical(updated) 

182 saved = cal.save_todo(ical) 

183 return self._to_task(saved, calendar_id=cal.name)

187 """Delete the task identified by *uid*. Idempotent. 

188 

189 When *calendar_id* is empty, iterates **all** calendars to locate 

190 the UID. Returns ``None`` when the UID does not exist in any 

191 calendar (already deleted). 

192 """ 

193 logger.debug("delete_task uid=%r calendar_id=%r", uid, calendar_id) 

194 if calendar_id: 

195 cal = self._get_calendar(calendar_id) 

196 try: 

197 task_obj = cal.get_todo_by_uid(uid) 

198 task_obj.delete() 

199 except self._caldav.lib.error.NotFoundError: 

200 return None 

201 else: 

202 result = self._find_task_by_uid(uid) 

203 if result is None: 

204 return None 

205 _cal, task_obj = result 

206 task_obj.delete() 

207 return None
